import React, { useState, useEffect } from 'react';
import { api } from './api';
import App from './App';
import './auth.css';
import { Shield, User } from 'lucide-react';

export default function AuthGate() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const [view, setView] = useState("login");
  
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [fullName, setFullName] = useState("");
  const [branch, setBranch] = useState("");
  const [academicYear, setAcademicYear] = useState("");
  const [passoutYear, setPassoutYear] = useState("");
  const [error, setError] = useState("");
  const [successMsg, setSuccessMsg] = useState("");

  useEffect(() => { checkAuth(); }, []);

  async function checkAuth() {
    try { setUser(await api.getMe()); } 
    catch (e) { setUser(null); } 
    finally { setLoading(false); }
  }

  async function handleLogin(e) {
    e.preventDefault();
    setError("");
    try {
      await api.login({ email, password });
      const me = await api.getMe();
      if (view === "hr_login" && me.role !== "company_hr") { setError("Not authorized for HR login."); await api.logout(); return; }
      if (view === "coordinator_login" && me.role !== "class_coordinator") { setError("Not authorized for Coordinator login."); await api.logout(); return; }
      setUser(me);
    } catch (err) { setError(err.message || "Invalid credentials"); }
  }

  async function handleSignup(e) {
    e.preventDefault();
    setError("");
    setSuccessMsg("");
    try {
      await api.register({ full_name: fullName, email, password, branch, academic_year: academicYear, graduation_year: passoutYear, role: "student" });
      setSuccessMsg("Account created! Please log in.");
      setView("login");
    } catch (err) { setError(err.message); }
  }

  if (loading) return <div className="auth-loading">Loading OnTrack...</div>;
  if (user) return <App user={user} onLogout={() => setUser(null)} />;

  return (
    <div className="auth-container">
      <div className="auth-card">
        <div className="auth-header">
          <h1>OnTrack</h1>
          <p>AI-Powered Student Analytics and Success Platform</p>
        </div>

        {error && <div className="error-banner">{error}</div>}
        {successMsg && <div className="success-banner">{successMsg}</div>}

        {view === "signup" ? (
          <form onSubmit={handleSignup} className="auth-form">
            <h3>Create Student Account</h3>
            <div className="input-group"><label>Full Name</label><input type="text" placeholder="Ayush Pandey" value={fullName} onChange={e => setFullName(e.target.value)} required /></div>
            <div className="input-group"><label>Branch</label><input type="text" placeholder="Computer Engineering" value={branch} onChange={e => setBranch(e.target.value)} required /></div>
            <div className="input-group"><label>Academic Year</label><input type="text" placeholder="3rd Year" value={academicYear} onChange={e => setAcademicYear(e.target.value)} required /></div>
            <div className="input-group"><label>Passout Year</label><input type="text" placeholder="2027" value={passoutYear} onChange={e => setPassoutYear(e.target.value)} required /></div>
            <div className="input-group"><label>Email Address</label><input type="email" placeholder="student@university.com" value={email} onChange={e => setEmail(e.target.value)} required /></div>
            <div className="input-group"><label>Password</label><input type="password" placeholder="••••••••" value={password} onChange={e => setPassword(e.target.value)} required /></div>
            <button type="submit" className="primary-btn">Register Account</button>
            <p className="switch-text">Already have an account? <span onClick={() => { setView("login"); setError(""); }}>Login here</span></p>
          </form>
        ) : (
          <form onSubmit={handleLogin} className="auth-form">
            <h3>{view === "hr_login" ? "Company HR Login" : view === "coordinator_login" ? "Class Coordinator Login" : "Student Login"}</h3>
            <div className="input-group"><label>Email Address</label><input type="email" placeholder="email@domain.com" value={email} onChange={e => setEmail(e.target.value)} required /></div>
            <div className="input-group"><label>Password</label><input type="password" placeholder="••••••••" value={password} onChange={e => setPassword(e.target.value)} required /></div>
            <button type="submit" className="primary-btn">{view === "hr_login" ? "Login as HR" : view === "coordinator_login" ? "Login as Coordinator" : "Login"}</button>
            {view === "login" ? (
              <p className="switch-text">Don't have an account? <span onClick={() => { setView("signup"); setError(""); }}>Create New Account</span></p>
            ) : (
              <p className="switch-text"><span onClick={() => { setView("login"); setError(""); }}>? Back to Student Login</span></p>
            )}
          </form>
        )}

        {view === "login" && (
          <div className="role-logins">
            <p>Other Portals:</p>
            <div className="role-buttons">
              <button type="button" className="role-btn" onClick={() => { setView("hr_login"); setError(""); }}><Shield size={14} /> Company HR</button>
              <button type="button" className="role-btn" onClick={() => { setView("coordinator_login"); setError(""); }}><User size={14} /> Class Coordinator</button>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}

import React, { useState, useEffect } from 'react';
import { api } from './api';
import './App.css';
import { LayoutDashboard, User, TrendingUp, Award, Target, Settings, LogOut, CheckCircle, Clock, Plus, Trash2, Save } from 'lucide-react';

export default function App({ user, onLogout }) {
  const [activeTab, setActiveTab] = useState("overview");
  const [dashboard, setDashboard] = useState(null);
  const [profile, setProfile] = useState({ phone: "", city: "", bio: "", college: "", degree: "", branch: "", academic_year: "", graduation_year: "", career_goal: "", preferred_role: "", courses: [], projects: [] });
  const [tasks, setTasks] = useState([]);
  const [sessions, setSessions] = useState([]);
  const [skills, setSkills] = useState([]);
  const [newSession, setNewSession] = useState({ session_date: "", topic: "", duration_minutes: 60, notes: "" });
  const [newSkill, setNewSkill] = useState({ skill_name: "", proficiency: "Beginner" });
  const [message, setMessage] = useState("");
  const [loading, setLoading] = useState(false);

  useEffect(() => { loadData(); }, [activeTab]);

  async function loadData() {
    try {
      if (user.role === "student") {
        setDashboard(await api.getDashboard());
        if (activeTab === "update_profile" || activeTab === "profile") {
          const p = await api.getProfile();
          if (p) setProfile(p);
        } else if (activeTab === "progress") {
          setTasks(await api.getTasks());
          setSessions(await api.getSessions());
        } else if (activeTab === "skills") {
          setSkills(await api.getSkills());
        }
      }
    } catch (e) { console.error(e); }
  }

  async function saveProfile(e) {
    e.preventDefault();
    setLoading(true);
    setMessage("");
    try {
      await api.saveProfile(profile);
      setMessage("Profile saved successfully!");
      loadData();
    } catch (err) { setMessage("Error: " + err.message); }
    finally { setLoading(false); }
  }

  return (
    <div className="dashboard-layout">
      <aside className="sidebar">
        <div className="sidebar-brand"><h2>OnTrack</h2><span>{user.role.replace("_", " ").toUpperCase()}</span></div>
        <nav className="sidebar-nav">
          {user.role === "student" && (
            <>
              <button className={activeTab === "overview" ? "active" : ""} onClick={() => setActiveTab("overview")}><LayoutDashboard size={18} /> Overview</button>
              <button className={activeTab === "update_profile" ? "active" : ""} onClick={() => setActiveTab("update_profile")}><User size={18} /> Profile Update</button>
              <button className={activeTab === "progress" ? "active" : ""} onClick={() => setActiveTab("progress")}><TrendingUp size={18} /> My Progress</button>
              <button className={activeTab === "skills" ? "active" : ""} onClick={() => setActiveTab("skills")}><Award size={18} /> Skills & Certifications</button>
              <button className={activeTab === "goals" ? "active" : ""} onClick={() => setActiveTab("goals")}><Target size={18} /> Career Goals & Jobs</button>
            </>
          )}
          <button className={activeTab === "settings" ? "active" : ""} onClick={() => setActiveTab("settings")}><Settings size={18} /> Settings</button>
          <button className="logout-btn" onClick={async () => { try { await api.logout(); } catch(e){} onLogout(); }}><LogOut size={18} /> Logout</button>
        </nav>
      </aside>

      <main className="main-content">
        <header className="top-header">
          <h1>Welcome, {user.full_name}</h1>
          <p className="role-badge">{user.email}</p>
        </header>

        <div className="content-body">
          {user.role !== "student" ? (
            <div className="card">
              <h3>{user.role === "company_hr" ? "Company HR Portal" : "Class Coordinator Portal"}</h3>
              <p>Portal access verified. Student tracking and internship management modules active.</p>
            </div>
          ) : (
            <>
              {activeTab === "overview" && dashboard && (
                <div className="overview-grid">
                  <div className="metric-card"><Clock className="metric-icon" /><div><h3>{dashboard.total_learning_hours} hrs</h3><p>Learning Hours</p></div></div>
                  <div className="metric-card"><CheckCircle className="metric-icon" /><div><h3>{dashboard.tasks_completed} / {dashboard.total_tasks}</h3><p>Tasks Completed</p></div></div>
                  <div className="metric-card"><TrendingUp className="metric-icon" /><div><h3>{dashboard.progress_percentage}%</h3><p>Progress Rate</p></div></div>
                  <div className="metric-card"><Award className="metric-icon" /><div><h3>{dashboard.skills_recorded}</h3><p>Skills Registered</p></div></div>
                </div>
              )}

              {activeTab === "update_profile" && (
                <div className="card">
                  <h2>Student Profile Update</h2>
                  {message && <div className="success-banner">{message}</div>}
                  <form onSubmit={saveProfile} className="profile-form">
                    <div className="form-grid">
                      <div><label>Phone Number</label><input type="text" value={profile.phone} onChange={e => setProfile({...profile, phone: e.target.value})} placeholder="+91 9876543210" /></div>
                      <div><label>City / Location</label><input type="text" value={profile.city} onChange={e => setProfile({...profile, city: e.target.value})} placeholder="Mumbai" /></div>
                      <div><label>College</label><input type="text" value={profile.college} onChange={e => setProfile({...profile, college: e.target.value})} placeholder="Universal SkillTech University" /></div>
                      <div><label>Branch</label><input type="text" value={profile.branch} onChange={e => setProfile({...profile, branch: e.target.value})} placeholder="Computer Engineering" /></div>
                      <div><label>Academic Year</label><input type="text" value={profile.academic_year} onChange={e => setProfile({...profile, academic_year: e.target.value})} placeholder="3rd Year" /></div>
                      <div><label>Passout Year</label><input type="text" value={profile.graduation_year} onChange={e => setProfile({...profile, graduation_year: e.target.value})} placeholder="2027" /></div>
                    </div>
                    <div className="input-group"><label>Bio / Introduction</label><textarea rows="3" value={profile.bio} onChange={e => setProfile({...profile, bio: e.target.value})} placeholder="Passionate engineering student..."></textarea></div>

                      </div>
                    ))}
                  </div>
                </div>
              )}

              {activeTab === "skills" && (
                <div className="card">
                  <h2>Skills & Certifications</h2>
                  <form onSubmit={async e => { e.preventDefault(); await api.createSkill(newSkill); setNewSkill({ skill_name: "", proficiency: "Beginner" }); loadData(); }} className="inline-form">
                    <input type="text" placeholder="Skill Name (e.g. React, Python)" value={newSkill.skill_name} onChange={e => setNewSkill({...newSkill, skill_name: e.target.value})} required />
                    <select value={newSkill.proficiency} onChange={e => setNewSkill({...newSkill, proficiency: e.target.value})}>
                      <option value="Beginner">Beginner</option>
                      <option value="Intermediate">Intermediate</option>
                      <option value="Advanced">Advanced</option>
                    </select>
                    <button type="submit" className="primary-btn"><Plus size={16} /> Add Skill</button>
                  </form>
                  <div className="skills-grid">
                    {skills.map(sk => (
                      <div key={sk.id} className="skill-chip">
                        <span><strong>{sk.skill_name}</strong> ({sk.proficiency})</span>
                        <button onClick={async () => { await api.deleteSkill(sk.id); loadData(); }} className="icon-btn"><Trash2 size={14} /></button>
                      </div>
                    ))}
                  </div>

const API_BASE = "http://localhost:8000/api";

async function request(endpoint, options = {}) {
  const config = {
    ...options,
    credentials: options.credentials ?? "include",
    headers: {
      "Content-Type": "application/json",
      ...options.headers,
    },
  };

  if (
    config.body &&
    typeof config.body === "object" &&
    !(config.body instanceof FormData)
  ) {
    config.body = JSON.stringify(config.body);
  }

  const response = await fetch(`${API_BASE}${endpoint}`, config);

  if (response.status === 204) return null;

  const data = await response.json().catch(() => ({
    detail: "Server error",
  }));

  if (!response.ok) {
    throw new Error(data.detail || "Something went wrong");
  }

  return data;
}

import psycopg
from fastapi import APIRouter, HTTPException, Response, Cookie
from pydantic import BaseModel, EmailStr
from typing import Optional
from app.db.database import get_database_url
from app.core.security import get_password_hash, verify_password, create_access_token, verify_access_token

router = APIRouter(prefix="/api/auth", tags=["Authentication"])

class UserRegister(BaseModel):
    full_name: str
    email: EmailStr
    password: str
    branch: Optional[str] = ""
    academic_year: Optional[str] = ""
    graduation_year: Optional[str] = ""
    role: Optional[str] = "student"

class UserLogin(BaseModel):
    email: EmailStr
    password: str

@router.post("/register")
def register(data: UserRegister):
    url = get_database_url()
    with psycopg.connect(url) as conn:
        with conn.cursor() as cur:
            cur.execute("SELECT id FROM users WHERE email = %s", (data.email,))
            if cur.fetchone():
                raise HTTPException(status_code=400, detail="Email already registered")
            
            hashed_pw = get_password_hash(data.password)
            cur.execute("INSERT INTO users (full_name, email, password_hash, role) VALUES (%s, %s, %s, %s) RETURNING id", 
                        (data.full_name, data.email, hashed_pw, data.role))
            user_id = cur.fetchone()[0]

            if data.role == "student":
                cur.execute("INSERT INTO student_profiles (user_id, branch, academic_year, graduation_year) VALUES (%s, %s, %s, %s) ON CONFLICT (user_id) DO NOTHING",
                            (user_id, data.branch, data.academic_year, data.graduation_year))
            conn.commit()
    return {"message": "Account created successfully"}

@router.post("/login")
def login(data: UserLogin, response: Response):
    url = get_database_url()
    with psycopg.connect(url) as conn:
        with conn.cursor() as cur:
            cur.execute("SELECT id, full_name, email, password_hash, role FROM users WHERE email = %s", (data.email,))
            user = cur.fetchone()
            if not user or not verify_password(data.password, user[3]):
                raise HTTPException(status_code=401, detail="Invalid email or password")
            
            token = create_access_token({"sub": str(user[0]), "role": user[4]})
            response.set_cookie(key="ontrack_session", value=token, httponly=True, max_age=604800, samesite="lax")
            return {"message": "Login successful", "role": user[4], "full_name": user[1]}

@router.get("/me")
def get_me(ontrack_session: str = Cookie(None)):
    if not ontrack_session:
        raise HTTPException(status_code=401, detail="Not authenticated")
    payload = verify_access_token(ontrack_session)
    if not payload:
        raise HTTPException(status_code=401, detail="Invalid session")
    
    url = get_database_url()
    with psycopg.connect(url) as conn:
        with conn.cursor() as cur:
            cur.execute("SELECT id, full_name, email, role FROM users WHERE id = %s", (payload.get("sub"),))
            user = cur.fetchone()
            if not user:
                raise HTTPException(status_code=401, detail="User not found")
            return {"id": user[0], "full_name": user[1], "email": user[2], "role": user[3]}

            cur.execute("SELECT COUNT(*) FROM student_skills WHERE user_id = %s", (user_id,))

            progress_pct = int((completed_tasks / total_tasks * 100)) if total_tasks > 0 else 0

                "total_tasks": total_tasks,
                "tasks_completed": completed_tasks,
                "skills_recorded": skills_count,
                "progress_percentage": progress_pct

import os

load_dotenv()

def get_database_url() -> str:
    db_url = os.getenv("DATABASE_URL")
    if not db_url:
        user = os.getenv("DB_USER", "postgres")
        password = os.getenv("DB_PASSWORD", "postgres")
        host = os.getenv("DB_HOST", "localhost")
        port = os.getenv("DB_PORT", "5432")
        dbname = os.getenv("DB_NAME", "ontrack")
        db_url = f"postgresql://{user}:{password}@{host}:{port}/{dbname}"
    return db_url

def create_tables():
    url = get_database_url()
    with psycopg.connect(url) as conn:
        with conn.cursor() as cur:
            cur.execute("""
                CREATE TABLE IF NOT EXISTS users (
                    id SERIAL PRIMARY KEY,
                    full_name VARCHAR(255) NOT NULL,
                    email VARCHAR(255) UNIQUE NOT NULL,
                    password_hash VARCHAR(255) NOT NULL,
                    role VARCHAR(50) NOT NULL DEFAULT 'student',
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                );
            """)
            cur.execute("""
                CREATE TABLE IF NOT EXISTS student_profiles (
                    id SERIAL PRIMARY KEY,
                    user_id INT UNIQUE REFERENCES users(id) ON DELETE CASCADE,
                    phone VARCHAR(50),
                    city VARCHAR(100),
                    bio TEXT,
                    college VARCHAR(255),
                    degree VARCHAR(255),
                    branch VARCHAR(255),
                    academic_year VARCHAR(50),
                    graduation_year VARCHAR(50),
                    career_goal TEXT,
                    preferred_role VARCHAR(255),
                    courses JSONB DEFAULT '[]'::jsonb,
                    projects JSONB DEFAULT '[]'::jsonb
                );
            """)
            cur.execute("""
                CREATE TABLE IF NOT EXISTS student_tasks (
                    id SERIAL PRIMARY KEY,
                    user_id INT REFERENCES users(id) ON DELETE CASCADE,
                    title VARCHAR(255) NOT NULL,
                    description TEXT,
                    due_date VARCHAR(50),
                    status VARCHAR(50) DEFAULT 'pending'
                );
            """)
            cur.execute("""
                CREATE TABLE IF NOT EXISTS learning_sessions (

                    topic VARCHAR(255) NOT NULL,

            """)

                CREATE TABLE IF NOT EXISTS student_skills (
                    id SERIAL PRIMARY KEY,
                    user_id INT REFERENCES users(id) ON DELETE CASCADE,
                    skill_name VARCHAR(100) NOT NULL,
                    proficiency VARCHAR(50) DEFAULT 'Beginner'
                );
            """)
